ALTER Procedure [dbo].[spAuthenticateadmin]
@UserName varchar(10),
@Password varchar(20)
as
Begin
 Declare @Count int
 
 Select @Count = COUNT(adm_username) from admin1
 where [adm_username] = @UserName and [adm_password] = @Password
 
 if(@Count = 1)
 Begin
  Select 1 as ReturnCode
 End
 Else
 Begin
  Select -1 as ReturnCode
 End
End
