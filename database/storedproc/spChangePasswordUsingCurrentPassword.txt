ALTER Proc [dbo].[spChangePasswordUsingCurrentPassword]
@UserName varchar(10),
@CurrentPassword varchar(20),
@NewPassword nvarchar(20)
as
Begin
if(Exists(Select f_name from customer 
 where customer_id = @UserName
 and [passwd] = @CurrentPassword))
Begin
Update customer
Set [passwd] = @NewPassword
where customer_id = @UserName

Select 1 as IsPasswordChanged
End
Else
Begin
Select 0 as IsPasswordChanged
End
End
