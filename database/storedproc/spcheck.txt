ALTER PROC [dbo].[spcheck] @UserName varchar(10)

AS
BEGIN
DECLARE @Count INT
DECLARE @zip bigint
DECLARE @ReturnCode INT

select @zip=zipcode from customer where customer_id=@UserName

SELECT @Count = count(zipcode)
FROM zip
WHERE zipcode = @zip
if @Count=1
Begin
  Select 1 as ReturnCode
 End
  Begin
  Select -1 as ReturnCode
 End

end
