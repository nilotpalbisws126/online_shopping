ALTER PROC [dbo].[spselect] @customer_id varchar(10)

AS
begin
DECLARE @Count INT
DECLARE @Cc INT
select @Count=count(product_id) from cart where customer_id=@customer_id
--if @Count>0
begin
select @cc=SUM(tot) from cart where customer_id=@customer_id 
select distinct customer_id,@cc as grndtot from cart where customer_id=@customer_id 
end
end
