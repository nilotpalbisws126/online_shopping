ALTER proc [dbo].[spResetPassword]
@UserName nvarchar(100)
as
Begin
 
 Declare @Email nvarchar(100)
 
 Select @Email = Email 
 from customer
 where customer_id = @UserName
 
 if(@Email IS NOT NULL)
 Begin
  --If username exists
  Declare @GUID UniqueIdentifier
  Set @GUID = NEWID()
  
  Insert into tblResetPasswordRequests
  (Id, UserId, ResetRequestDateTime)
  Values(@GUID, @UserName, GETDATE())
  
  Select 1 as ReturnCode, @GUID as UniqueId, @Email as Email
 End
 Else
 Begin
  --If username does not exist
  SELECT 0 as ReturnCode, NULL as UniqueId, NULL as Email
 End
End
